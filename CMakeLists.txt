cmake_minimum_required(VERSION 3.20)
project(rudy_netproof VERSION 0.1.0 LANGUAGES CXX)
option(NETPROOF_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(NETPROOF_REQUIRE_CLI_TESTS "Require Python for the independent CLI contract suite" OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(netproof_core src/model.cpp src/engine.cpp src/output.cpp)
target_include_directories(netproof_core PUBLIC include)
target_include_directories(netproof_core SYSTEM PUBLIC third_party)
target_compile_features(netproof_core PUBLIC cxx_std_20)
add_executable(netproof src/main.cpp)
target_link_libraries(netproof PRIVATE netproof_core)
add_executable(netproof_tests tests/tests.cpp)
target_link_libraries(netproof_tests PRIVATE netproof_core)
foreach(target netproof_core netproof netproof_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
    if(MINGW)
      target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
    elseif(UNIX AND NOT APPLE AND NOT NETPROOF_SANITIZE)
      target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
    endif()
    if(NETPROOF_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME core-regressions-and-exhaustive-oracle COMMAND netproof_tests)
add_test(NAME cli-help COMMAND netproof --help)
add_test(NAME cli-version COMMAND netproof --version)
if(NETPROOF_REQUIRE_CLI_TESTS)
  find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
else()
  find_package(Python3 3.9 COMPONENTS Interpreter)
endif()
if(Python3_FOUND)
  add_test(NAME cli-contracts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py $<TARGET_FILE:netproof> ${CMAKE_CURRENT_SOURCE_DIR})
endif()
install(TARGETS netproof RUNTIME DESTINATION bin)
install(FILES README.md LICENSE SECURITY.md THIRD_PARTY.md DESTINATION .)
install(DIRECTORY docs examples assets DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses RENAME nlohmann-json-MIT.txt)
